cmake_minimum_required(VERSION 3.25)
project(SerialLinkLifecycleTest LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.8 REQUIRED COMPONENTS Core SerialPort Test)
get_filename_component(root "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
set(serial_source "${root}/src/Comms/SerialLink.cc")
set(serial_header "${root}/src/Comms/SerialLink.h")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${serial_source}" "${serial_header}")
file(READ "${serial_source}" serial)
file(READ "${serial_header}" header)
string(FIND "${serial}" "QGC_LOGGING_CATEGORY(SerialLinkLog" begin)
if(begin LESS 0)
    message(FATAL_ERROR "SerialLink production boundary not found")
endif()
string(SUBSTRING "${serial}" ${begin} -1 production)
string(REPLACE "#include \"LinkConfiguration.h\"" "" header "${header}")
string(REPLACE "#include \"LinkInterface.h\"" "" header "${header}")
# Expose the private factory entry only in the generated test declaration.
string(REPLACE "private:" "public:" header "${header}")
configure_file(Fixture.h.in Fixture.h @ONLY)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/Production.cc" "#include \"Fixture.h\"\n${production}\n")
add_executable(SerialLinkLifecycleTest SerialLinkLifecycleTest.cc
    "${CMAKE_CURRENT_BINARY_DIR}/Fixture.h" "${CMAKE_CURRENT_BINARY_DIR}/Production.cc"
    "${root}/src/Comms/HarmonySerialBridge.cc" "${root}/src/Comms/HarmonySerialBridge.h")
set_source_files_properties(SerialLinkLifecycleTest.cc "${CMAKE_CURRENT_BINARY_DIR}/Production.cc"
    PROPERTIES COMPILE_DEFINITIONS Q_OS_HARMONY)
target_compile_definitions(SerialLinkLifecycleTest PRIVATE QGC_UNITTEST_BUILD)
target_include_directories(SerialLinkLifecycleTest PRIVATE "${CMAKE_CURRENT_BINARY_DIR}" "${root}/src/Comms")
target_link_libraries(SerialLinkLifecycleTest PRIVATE Qt6::Core Qt6::SerialPort Qt6::Test)
enable_testing()
add_test(NAME SerialLinkLifecycleTest COMMAND SerialLinkLifecycleTest
    -o "${CMAKE_CURRENT_BINARY_DIR}/result.txt,txt" -o "${CMAKE_CURRENT_BINARY_DIR}/result.xml,junitxml")
set_tests_properties(SerialLinkLifecycleTest PROPERTIES TIMEOUT 15)
