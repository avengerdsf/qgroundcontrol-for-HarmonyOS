get_filename_component(repository "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
function(section source first last output)
    string(FIND "${source}" "${first}" start)
    if(start LESS 0)
        message(FATAL_ERROR "Missing production section: ${first}")
    endif()
    string(SUBSTRING "${source}" ${start} -1 remainder)
    string(FIND "${remainder}" "${last}" end)
    if(end LESS 0)
        message(FATAL_ERROR "Missing production boundary: ${last}")
    endif()
    string(SUBSTRING "${remainder}" 0 ${end} value)
    set(${output} "${value}" PARENT_SCOPE)
endfunction()
set(controller_header "${repository}/src/Vehicle/VehicleSetup/FirmwareUpgradeController.h")
set(controller_source "${repository}/src/Vehicle/VehicleSetup/FirmwareUpgradeController.cc")
set(info_source "${repository}/src/Comms/QGCSerialPortInfo.cc")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${controller_header}" "${controller_source}" "${info_source}")
file(READ "${controller_header}" header)
file(READ "${controller_source}" controller)
file(READ "${info_source}" info)
section("${header}" "typedef enum {" "    FirmwareUpgradeController(void);" enums)
string(REGEX MATCH "void[ \t]+_foundBoard[ \t]*\\([^;]+;" found_declaration "${header}")
string(REGEX MATCH "(QGCSerialPortInfo|QSerialPortInfo)[ \t]+_boardInfo;" board_member "${header}")
string(REGEX MATCH "QString boardPort[^\n]+" port_getter "${header}")
string(REGEX MATCH "QString boardDescription[^\n]+" description_getter "${header}")
string(REGEX MATCH "typedef struct \\{[^}]+\\} ManifestFirmwareInfo_t;" manifest "${header}")
foreach(item found_declaration board_member port_getter description_getter manifest)
    if(NOT ${item})
        message(FATAL_ERROR "Missing production declaration: ${item}")
    endif()
endforeach()
set(properties "")
foreach(name boardPort boardDescription apmFirmwareNames apmFirmwareNamesBestIndex apmFirmwareUrls)
    string(REGEX MATCH "Q_PROPERTY\\([^\n]*[ \t]${name}[ \t][^\n]*" property "${header}")
    if(NOT property)
        message(FATAL_ERROR "Missing production property: ${name}")
    endif()
    string(APPEND properties "${property}\n")
endforeach()
