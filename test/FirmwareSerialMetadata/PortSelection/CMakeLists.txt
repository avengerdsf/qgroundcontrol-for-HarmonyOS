cmake_minimum_required(VERSION 3.25)
project(FirmwarePortSelectionTest LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Test)
get_filename_component(root "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
set(worker_file "${root}/src/Vehicle/VehicleSetup/PX4FirmwareUpgradeThread.cc")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${worker_file}")
file(READ "${worker_file}" worker)
function(section first last output)
    string(FIND "${worker}" "${first}" begin)
    if(begin LESS 0)
        message(FATAL_ERROR "Production worker boundary missing: ${first}")
    endif()
    string(SUBSTRING "${worker}" ${begin} -1 remainder)
    string(FIND "${remainder}" "${last}" end)
    if(end LESS 0)
        message(FATAL_ERROR "Production worker boundary missing: ${last}")
    endif()
    string(SUBSTRING "${remainder}" 0 ${end} result)
    set(${output} "${result}" PARENT_SCOPE)
endfunction()
section("void PX4FirmwareUpgradeThreadWorker::_resumeDiscovery(" "void PX4FirmwareUpgradeThreadWorker::_setTargetPort(" resume)
section("void PX4FirmwareUpgradeThreadWorker::_setTargetPort(" "void PX4FirmwareUpgradeThreadWorker::_reboot(" selection)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/Production.cc"
    "#include \"Fixture.h\"\nQ_LOGGING_CATEGORY(PX4FirmwareUpgradeThreadLog, \"test.firmware\")\nQ_LOGGING_CATEGORY(PX4FirmwareUpgradeThreadVerboseLog, \"test.firmware.verbose\")\n${resume}\n${selection}\n")
configure_file(Fixture.h.in Fixture.h COPYONLY)
add_executable(FirmwarePortSelectionTest FirmwarePortSelectionTest.cc "${CMAKE_CURRENT_BINARY_DIR}/Fixture.h"
    "${CMAKE_CURRENT_BINARY_DIR}/Production.cc")
target_include_directories(FirmwarePortSelectionTest PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_compile_definitions(FirmwarePortSelectionTest PRIVATE Q_OS_HARMONY)
target_link_libraries(FirmwarePortSelectionTest PRIVATE Qt6::Core Qt6::Test)
enable_testing()
add_test(NAME FirmwarePortSelectionTest COMMAND FirmwarePortSelectionTest
    -o "${CMAKE_CURRENT_BINARY_DIR}/result.txt,txt" -o "${CMAKE_CURRENT_BINARY_DIR}/result.xml,junitxml")
set_tests_properties(FirmwarePortSelectionTest PROPERTIES TIMEOUT 15)
