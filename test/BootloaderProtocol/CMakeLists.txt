cmake_minimum_required(VERSION 3.25)
project(BootloaderProtocolTest LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.12 REQUIRED COMPONENTS Core Test)
get_filename_component(repository "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
function(section source first last output)
    string(FIND "${source}" "${first}" start)
    if(start LESS 0)
        message(FATAL_ERROR "Missing production section: ${first}")
    endif()
    string(SUBSTRING "${source}" ${start} -1 remainder)
    string(FIND "${remainder}" "${last}" end)
    if(end LESS 0)
        message(FATAL_ERROR "Missing production boundary: ${last}")
    endif()
    string(SUBSTRING "${remainder}" 0 ${end} result)
    set(${output} "${result}" PARENT_SCOPE)
endfunction()
set(boot_source "${repository}/src/Vehicle/VehicleSetup/Bootloader.cc")
set(boot_header "${repository}/src/Vehicle/VehicleSetup/Bootloader.h")
set(crc_source "${repository}/src/Utilities/QGC.cc")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${boot_source}" "${boot_header}" "${crc_source}")
file(READ "${boot_source}" boot)
file(READ "${boot_header}" header)
file(READ "${crc_source}" crc)
section("${boot}" "bool Bootloader::_binProgram(" "bool Bootloader::_ihxProgram(" programming)
section("${boot}" "bool Bootloader::_write(const uint8_t* data" "bool Bootloader::_read(" writing)
section("${boot}" "bool Bootloader::_getCommandResponse(" "bool Bootloader::_protoGetDevice(" response)
section("${header}" "    enum {" "\n#ifdef Q_OS_HARMONY" constants)
section("${crc}" "static const quint32 crctab[]" "bool fuzzyCompare(" checksum)
configure_file(Fixture.h.in "${CMAKE_CURRENT_BINARY_DIR}/Fixture.h" @ONLY)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/Production.cc"
    "#include \"Fixture.h\"\nnamespace QGC {\n${checksum}\n}\n${writing}\n${response}\n${programming}\n")
add_executable(BootloaderProtocolTest BootloaderProtocolTest.cc
    "${CMAKE_CURRENT_BINARY_DIR}/Fixture.h" "${CMAKE_CURRENT_BINARY_DIR}/Production.cc")
target_include_directories(BootloaderProtocolTest PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_compile_definitions(BootloaderProtocolTest PRIVATE Q_OS_HARMONY)
target_link_libraries(BootloaderProtocolTest PRIVATE Qt6::Core Qt6::Test)
enable_testing()
add_test(NAME BootloaderProtocolTest COMMAND BootloaderProtocolTest
    -o "${CMAKE_CURRENT_BINARY_DIR}/result.txt,txt" -o "${CMAKE_CURRENT_BINARY_DIR}/result.xml,junitxml")
set_tests_properties(BootloaderProtocolTest PROPERTIES TIMEOUT 15)
