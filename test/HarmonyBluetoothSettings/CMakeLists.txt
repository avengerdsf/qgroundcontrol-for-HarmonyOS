cmake_minimum_required(VERSION 3.25)
project(HarmonyBluetoothSettingsTest LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Core Gui Qml Quick Test)
foreach(pair "QGCButton|Button" "QGCLabel|Label" "QGCCheckBox|CheckBox" "QGCRadioButton|RadioButton" "SectionHeader|Label" "QGCFlickable|Flickable" "QGCCheckBoxSlider|CheckBox")
    string(REPLACE "|" ";" fields "${pair}")
    list(GET fields 0 name)
    list(GET fields 1 base)
    file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/${name}.qml" "import QtQuick\nimport QtQuick.Controls\n${base} {}\n")
endforeach()
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/QGCComboBox.qml" "import QtQuick.Controls\nComboBox { property bool sizeToContents: false }\n")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/QGCTextField.qml" "import QtQuick.Controls\nTextField { property bool validationError: false }\n")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/ScreenTools.qml" "pragma Singleton\nimport QtQml\nQtObject { property real defaultFontPixelWidth: 8; property real defaultFontPixelHeight: 16; property real smallFontPointSize: 10; property bool isSerialAvailable: false; property real minTouchPixels: 24 }\n")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/QGCPalette.qml" "import QtQuick\nQtObject { property bool colorGroupEnabled: true; property color text: 'black'; property color warningText: 'red'; property color buttonText: 'gray'; property color buttonHighlightText: 'white' }\n")
add_executable(HarmonyBluetoothSettingsTest BluetoothSettingsTest.cc)
target_link_libraries(HarmonyBluetoothSettingsTest PRIVATE Qt6::Core Qt6::Gui Qt6::Qml Qt6::Quick Qt6::Test)
target_compile_definitions(HarmonyBluetoothSettingsTest PRIVATE PRODUCT_QML_PATH="${CMAKE_CURRENT_LIST_DIR}/../../src/AppSettings/BluetoothSettings.qml" TEST_BINARY_DIR="${CMAKE_CURRENT_BINARY_DIR}")
enable_testing()
add_test(NAME HarmonyBluetoothSettingsTest COMMAND HarmonyBluetoothSettingsTest -o "${CMAKE_CURRENT_BINARY_DIR}/result.txt,txt" -o "${CMAKE_CURRENT_BINARY_DIR}/result.xml,junitxml")

file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/SettingsGroupLayout.qml" "import QtQuick.Layouts\nColumnLayout { property string heading }\n")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/LabelledButton.qml" "import QtQuick.Controls\nButton { property string label; property string buttonText }\n")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/QGCColoredImage.qml" "import QtQuick\nImage { property color color }\n")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/QGCMouseArea.qml" "import QtQuick\nMouseArea { property Item fillItem }\n")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/QGCPopupDialog.qml" "import QtQuick\nItem { property string title; property var buttons; property bool acceptButtonEnabled: true; property bool preventClose: false; signal accepted(); signal rejected(); function open() {} }\n")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/LabelledComboBox.qml" "import QtQuick\nimport QtQuick.Controls\nItem { property string label; property var model; property alias comboBox: combo; signal activated(int index); ComboBox { id: combo } }\n")
foreach(kind "Legacy" "Invalid" "RejectSave" "Valid" "NoSaveMethod")
    set(body "")
    if(kind STREQUAL "Invalid")
        set(body "property bool settingsValid: false; function saveSettings() { return true }")
    elseif(kind STREQUAL "RejectSave")
        set(body "property bool settingsValid: true; function saveSettings() { return false }")
    elseif(kind STREQUAL "Valid")
        set(body "property bool settingsValid: true; function saveSettings() { return true }")
    elseif(kind STREQUAL "Legacy")
        set(body "function saveSettings() {}")
    endif()
    file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/${kind}LinkSettings.qml" "import QtQuick\nItem { ${body} }\n")
endforeach()
target_compile_definitions(HarmonyBluetoothSettingsTest PRIVATE
    PRODUCT_PARENT_QML_PATH="${CMAKE_CURRENT_LIST_DIR}/../../src/AppSettings/LinkConfigurationManager.qml"
    PRODUCT_FACTORY_QML_PATH="${CMAKE_CURRENT_LIST_DIR}/../../src/QmlControls/QGCPopupDialogFactory.qml")
