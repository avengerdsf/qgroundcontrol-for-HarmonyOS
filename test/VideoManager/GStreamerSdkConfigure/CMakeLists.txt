cmake_minimum_required(VERSION 3.25)
project(GStreamerSdkConfigureTest LANGUAGES NONE)

if(NOT OHOS_ARCH OR NOT CMAKE_SYSROOT)
    message(FATAL_ERROR "Set OHOS_ARCH and CMAKE_SYSROOT for the real OHOS target SDK")
endif()
set(QGC_PLATFORM_OHOS ON)
set(QGC_BUILD_TESTING ON)
if(OHOS_ARCH STREQUAL "arm64-v8a")
    set(CMAKE_SYSTEM_PROCESSOR aarch64)
else()
    set(CMAKE_SYSTEM_PROCESSOR x86_64)
endif()
get_filename_component(_qgc_root "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
list(PREPEND CMAKE_MODULE_PATH "${_qgc_root}/cmake/find-modules")

# A host installation must not satisfy any dependency of the target SDK.
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/host-pkgconfig")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/host-pkgconfig/gstreamer-1.0.pc"
    "Name: host-gstreamer\nDescription: host isolation sentinel\nVersion: 99.0.0\nLibs: -lhost_gstreamer_must_not_be_used\n")
set(ENV{PKG_CONFIG_PATH} "${CMAKE_CURRENT_BINARY_DIR}/host-pkgconfig")
set(ENV{PKG_CONFIG_LIBDIR} "${CMAKE_CURRENT_BINARY_DIR}/host-pkgconfig")
find_package(GStreamer 1.28.4 EXACT REQUIRED COMPONENTS Core Base Video Gl GlPrototypes GlEgl Rtsp App RtspServer)

if(NOT GStreamer_VERSION VERSION_EQUAL 1.28.4 OR NOT GStreamer_USE_STATIC_LIBS)
    message(FATAL_ERROR "The real target SDK was not selected")
endif()
get_target_property(_libraries GStreamer::GStreamer INTERFACE_LINK_LIBRARIES)
foreach(required libgstreamer-1.0.a libgstopengl.a libgstlibav.a libgioopenssl.a libgstrtspserver-1.0.a)
    string(FIND "${_libraries}" "${required}" _found)
    if(_found EQUAL -1)
        message(FATAL_ERROR "Missing real static target dependency: ${required}")
    endif()
endforeach()
foreach(library IN LISTS _libraries)
    if(IS_ABSOLUTE "${library}")
        cmake_path(IS_PREFIX GStreamer_ROOT_DIR "${library}" NORMALIZE _in_prefix)
        cmake_path(IS_PREFIX CMAKE_SYSROOT "${library}" NORMALIZE _in_sysroot)
        if(NOT _in_prefix AND NOT _in_sysroot)
            message(FATAL_ERROR "Host dependency leaked into the target: ${library}")
        endif()
    endif()
endforeach()
foreach(property INTERFACE_LINK_OPTIONS INTERFACE_COMPILE_OPTIONS)
    get_target_property(_options GStreamer::GStreamer ${property})
    if(_options)
        foreach(flag IN LISTS _options)
            if(flag STREQUAL "-pthread"
                    OR (property STREQUAL "INTERFACE_LINK_OPTIONS" AND flag STREQUAL "-Wl,--export-dynamic")
                    OR (property STREQUAL "INTERFACE_COMPILE_OPTIONS" AND flag MATCHES "^-D[A-Za-z_][A-Za-z0-9_]*(=[A-Za-z0-9_]+)?$"))
                continue()
            endif()
            message(FATAL_ERROR "Unchecked OTHER flag in target options: ${flag}")
        endforeach()
    endif()
endforeach()
if(NOT "$ENV{PKG_CONFIG_PATH}" STREQUAL "${CMAKE_CURRENT_BINARY_DIR}/host-pkgconfig")
    message(FATAL_ERROR "FindGStreamer did not restore the caller's pkg-config environment")
endif()
message(STATUS "PASS: real OHOS GStreamer 1.28.4 static SDK, GL/EGL, TLS, test codecs and host isolation")
