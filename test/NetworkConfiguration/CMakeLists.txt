project(NetworkConfigurationTest LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.12 REQUIRED COMPONENTS Core Network Test)

function(read_section path first last output)
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${path}")
    file(READ "${path}" source)
    string(FIND "${source}" "${first}" start)
    if(start LESS 0)
        message(FATAL_ERROR "Missing production section: ${first}")
    endif()
    string(SUBSTRING "${source}" ${start} -1 remainder)
    string(FIND "${remainder}" "${last}" end)
    if(end LESS 0)
        message(FATAL_ERROR "Missing production boundary: ${last}")
    endif()
    string(SUBSTRING "${remainder}" 0 ${end} section)
    set(${output} "${section}" PARENT_SCOPE)
endfunction()

set(comms "${CMAKE_CURRENT_LIST_DIR}/../../src/Comms")
read_section("${comms}/UDPLink.h" "struct UDPClient" "class UDPWorker :" udp_header)
read_section("${comms}/TCPLink.h" "class TCPConfiguration :" "class TCPWorker :" tcp_header)
read_section("${comms}/UDPLink.cc" "namespace {" "const QHostAddress UDPWorker::_multicastGroup" udp_source)
read_section("${comms}/TCPLink.cc" "TCPConfiguration::TCPConfiguration" "TCPWorker::TCPWorker" tcp_source)
configure_file(Configuration.h.in Configuration.h @ONLY)
configure_file(Configuration.cc.in Configuration.cc @ONLY)

add_executable(NetworkConfigurationTest ConfigurationTest.cc
    "${CMAKE_CURRENT_BINARY_DIR}/Configuration.h" "${CMAKE_CURRENT_BINARY_DIR}/Configuration.cc")
target_include_directories(NetworkConfigurationTest PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_compile_definitions(NetworkConfigurationTest PRIVATE TEST_BINARY_DIR="${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(NetworkConfigurationTest PRIVATE Qt6::Core Qt6::Network Qt6::Test)

enable_testing()
add_test(NAME NetworkConfigurationTest COMMAND NetworkConfigurationTest
    -o "${CMAKE_CURRENT_BINARY_DIR}/result.txt,txt"
    -o "${CMAKE_CURRENT_BINARY_DIR}/result.xml,junitxml")
set_tests_properties(NetworkConfigurationTest PROPERTIES TIMEOUT 30)
